```c++
void Session::Send(SendBufferRef sendBuffer)
{
	if (IsConnected() == false)
		return;

	bool registerSend = false;

	// 현재 RegisterSend가 걸리지 않은 상태라면, 걸어준다
	{
		WRITE_LOCK;

		_sendQueue.push(sendBuffer);

		// 한 번만 값을 변경하기 위함
		if (_sendRegistered.exchange(true) == false)
			registerSend = true;
	}

	if (registerSend)
		RegisterSend();
}
```
{% endraw %}


- WRITE_LOCK : 쓰기 락을 설정하여, 다중 스레드 환경에서 `_sendQueue`에 대한 동시 접근을 방지
- `_sendRegistered`: 전송이 이미 등록되어 있는 지를 나타내는 플래그
- RegisterSend() : 실제로 데이터를 전송하기 위해 시스템에 전송 작업을 등록

```c++
void Session::Dispatch(IocpEvent* iocpEvent, int32 numOfBytes)
{
	switch (iocpEvent->eventType)
	{
	case EventType::Connect:
		ProcessConnect();
		break;
	case EventType::Disconnect:
		ProcessDisconnect();
		break;
	case EventType::Recv:
		ProcessRecv(numOfBytes);
		break;
	case EventType::Send:
		ProcessSend(numOfBytes);
		break;
	default:
		break;
	}
}
```

```c++
bool Session::RegisterConnect()
{
  // 이미 연결되어있는지 확인
	if (IsConnected())
		return false;

  // 클라이언트인지 확인
	if (GetService()->GetServiceType() != ServiceType::Client)
		return false;

	// 소켓 재사용 옵션 설정, 설정 실패 시 false 반환
	if (SocketUtils::SetReuseAddress(_socket, true) == false)
		return false;

  // 소켓을 임의의 주소에 바인딩, 실패 시 false 반환
	if (SocketUtils::BindAnyAddress(_socket, 0/*남는거*/) == false)
		return false;

  // 연결 이벤트를 초기화
	_connectEvent.Init();
	// 이벤트의 소유자를 현재 객체로 설정 (참조 카운트를 증가시킴)
	_connectEvent.owner = shared_from_this(); // ADD_REF

	DWORD numOfBytes = 0;
	// 연결할 서버의 주소를 가져옴
	SOCKADDR_IN sockAddr = GetService()->GetNetAddress().GetSockAddr();
	// 비동기 연결을 시도
	if (false == SocketUtils::ConnectEx(_socket, reinterpret_cast<SOCKADDR*>(&sockAddr), sizeof(sockAddr), nullptr, 0, &numOfBytes, &_connectEvent))
	{
		int32 errorCode = ::WSAGetLastError();
		if (errorCode != WSA_IO_PENDING)
		{
			// 실패 시 참조 카운트 감소시킴
			_connectEvent.owner = nullptr; // RELEASE_REF
			return false;
		}
	}
	// 작업이 성공적으로 완료되면 true 반환
	return true;
}
```

```c++
bool Session::RegisterDisconnect()
{
	_disconnectEvent.Init();
	_disconnectEvent.owner = shared_from_this(); // ADD_REF

	if (false == SocketUtils::DisconnectEx(_socket, &_disconnectEvent, TF_REUSE_SOCKET, 0))
	{
		int32 errorCode = ::WSAGetLastError();
		if (errorCode != WSA_IO_PENDING)
		{
			_disconnectEvent.owner = nullptr; // RELEASE_REF
			return false;
		}
	}

	return true;
}
```
{% endraw %}


- `_disconnectEvent` 객체의 `owner`를 현재 `Session` 객체로 설정
	- `shared_from_this()`는 현재 객체의 `shared_ptr`을 반환하여 참조 카운트를 증가시킴
- `SocketUtils::DisconnectEx` 함수를 호출하여 비동기적으로 소켓 연결 해제를 요청
	- 성공하면 `true`, 실패하면 `false`를 반환
- 오류 코드가 `WSA_IO_PENDING`인지 확인
	- `WSA_IO_PENDING`은 비동기 작업이 성공적으로 시작되었음을 의미
- 만약 오류 코드가 `WSA_IO_PENDING`이 아니라면, `owner`를 `nullptr`로 설정하여 참조 해제

```c++
void Session::RegisterRecv()
{
	// 이미 연결되어있는지 확인
	if (IsConnected() == false)
		return;

	_recvEvent.Init();
	// _recvEvent의 소유자를 현재 세션 객체로 설정
	_recvEvent.owner = shared_from_this(); // ADD_REF

	// 수신 데이터를 저장할 버퍼를 설정
	WSABUF wsaBuf;
	// 데이터가 기록될 위치
	wsaBuf.buf = reinterpret_cast<char*>(_recvBuffer.WritePos());
	// 버퍼의 남은 크기
	wsaBuf.len = _recvBuffer.FreeSize();

	DWORD numOfBytes = 0;
	DWORD flags = 0;
	// 비동기 수신 작업 시작
	if (SOCKET_ERROR == ::WSARecv(_socket, &wsaBuf, 1, OUT &numOfBytes, OUT &flags, &_recvEvent, nullptr))
	{
		int32 errorCode = ::WSAGetLastError();
		if (errorCode != WSA_IO_PENDING)
		{
			HandleError(errorCode);
			_recvEvent.owner = nullptr; // RELEASE_REF
		}
	}
}
```

```c++
void Session::RegisterSend()
{
	if (IsConnected() == false)
		return;

	_sendEvent.Init();
	_sendEvent.owner = shared_from_this(); // ADD_REF

	// 보낼 데이터를 sendEvent에 등록
	{
		WRITE_LOCK;

		int32 writeSize = 0;
		// _sendQueue가 비어있지 않은 동안 반복하여 데이터를 꺼내고,
		// writeSize를 갱신하며, sendBuffers에 추가
		while (_sendQueue.empty() == false)
		{
			SendBufferRef sendBuffer = _sendQueue.front();

			writeSize += sendBuffer->WriteSize();
			// TODO : 예외 체크

			_sendQueue.pop();
			_sendEvent.sendBuffers.push_back(sendBuffer);
		}
	}

	// Scatter-Gather (흩어져 있는 데이터들을 모아서 한 방에 보낸다)
	// WSABUF 구조체를 사용하여 데이터를 준비
	Vector<WSABUF> wsaBufs;
	// _sendEvent.sendBuffers의 크기만큼 벡터의 용량을 미리 할당
	// 메모리 할당을 최적화
	wsaBufs.reserve(_sendEvent.sendBuffers.size());
	// _sendEvent.sendBuffers에 있는 각 SendBufferRef 객체에 대해 반복
	for (SendBufferRef sendBuffer : _sendEvent.sendBuffers)
	{
	  // _sendEvent.sendBuffers에 저장된 각 sendBuffer에 대해 WSABUF를 생성
		WSABUF wsaBuf;
		// sendBuffer의 Buffer() 함수가 반환하는 포인터를 char* 타입으로 변환하여 wsaBuf.buf에 저장
		wsaBuf.buf = reinterpret_cast<char*>(sendBuffer->Buffer());
		wsaBuf.len = static_cast<LONG>(sendBuffer->WriteSize());
		wsaBufs.push_back(wsaBuf);
	}

	DWORD numOfBytes = 0;
	if (SOCKET_ERROR == ::WSASend(_socket, wsaBufs.data(), static_cast<DWORD>(wsaBufs.size()), OUT &numOfBytes, 0, &_sendEvent, nullptr))
	{
		int32 errorCode = ::WSAGetLastError();
		if (errorCode != WSA_IO_PENDING)
		{
			HandleError(errorCode);
			_sendEvent.owner = nullptr; // RELEASE_REF
			_sendEvent.sendBuffers.clear(); // RELEASE_REF
			_sendRegistered.store(false);
		}
	}
}
```

```c++
void Session::ProcessConnect()
{
	_connectEvent.owner = nullptr; // RELEASE_REF

	_connected.store(true);

	// 세션 등록
	GetService()->AddSession(GetSessionRef());

	// 컨텐츠 코드에서 재정의
	OnConnected();

	// 수신 등록
	RegisterRecv();
}
```

```c++
void Session::ProcessRecv(int32 numOfBytes)
{
	_recvEvent.owner = nullptr; // RELEASE_REF

	if (numOfBytes == 0)
	{
		Disconnect(L"Recv 0");
		return;
	}

	if (_recvBuffer.OnWrite(numOfBytes) == false)
	{
		Disconnect(L"OnWrite Overflow");
		return;
	}

	int32 dataSize = _recvBuffer.DataSize();
	int32 processLen = OnRecv(_recvBuffer.ReadPos(), dataSize); // 컨텐츠 코드에서 재정의
	if (processLen < 0 || dataSize < processLen || _recvBuffer.OnRead(processLen) == false)
	{
		Disconnect(L"OnRead Overflow");
		return;
	}

	// 커서 정리
	_recvBuffer.Clean();

	// 수신 등록
	RegisterRecv();
}
```

```c++
void Session::ProcessSend(int32 numOfBytes)
{
	_sendEvent.owner = nullptr; // RELEASE_REF
	_sendEvent.sendBuffers.clear(); // RELEASE_REF

	if (numOfBytes == 0)
	{
		Disconnect(L"Send 0");
		return;
	}

	// 컨텐츠 코드에서 재정의
	OnSend(numOfBytes);

	WRITE_LOCK;
	if (_sendQueue.empty())
		_sendRegistered.store(false);
	else
		RegisterSend();
}
```

```c++
void Session::HandleError(int32 errorCode)
{
	switch (errorCode)
	{
	case WSAECONNRESET:
	case WSAECONNABORTED:
		Disconnect(L"HandleError");
		break;
	default:
		// TODO : Log
		cout << "Handle Error : " << errorCode << endl;
		break;
	}
}
```
